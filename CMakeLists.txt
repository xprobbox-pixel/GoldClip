cmake_minimum_required(VERSION 3.24)
include(cmake/Identity.cmake)
project(AurumClip VERSION ${AURUM_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AURUM_BUILD_PLUGINS "Build JUCE native plugins and host integration tests" ON)
include(CTest)
add_library(aurum_dsp STATIC src/dsp/ProcessingEngine.cpp)
target_include_directories(aurum_dsp PUBLIC src)
set_target_properties(aurum_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(aurum_dsp PRIVATE /W4)
else()
 target_compile_options(aurum_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
 add_executable(AurumDspTests tests/DspTests.cpp)
 target_link_libraries(AurumDspTests PRIVATE aurum_dsp)
 add_test(NAME dsp-foundation COMMAND AurumDspTests)
endif()
if(NOT AURUM_BUILD_PLUGINS)
 return()
endif()
set(JUCE_SOURCE_DIR "" CACHE PATH "Local JUCE checkout at the pinned commit")
if(JUCE_SOURCE_DIR)
 execute_process(COMMAND git -C "${JUCE_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE juce_revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE juce_git_result)
 if(NOT juce_git_result EQUAL 0 OR NOT juce_revision STREQUAL AURUM_JUCE_REV)
  message(FATAL_ERROR "JUCE_SOURCE_DIR must be a Git checkout at ${AURUM_JUCE_REV}")
 endif()
 add_subdirectory("${JUCE_SOURCE_DIR}" juce)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG ${AURUM_JUCE_REV})
 FetchContent_MakeAvailable(JUCE)
endif()
function(aurum_variant target product code suffix compact)
 set(formats VST3 Standalone)
 if(APPLE)
  list(APPEND formats AU)
 endif()
 juce_add_plugin(${target} COMPANY_NAME "${AURUM_MANUFACTURER}" BUNDLE_ID "${AURUM_BUNDLE_PREFIX}.${suffix}"
  PLUGIN_MANUFACTURER_CODE ${AURUM_MANUFACTURER_CODE} PLUGIN_CODE ${code} PRODUCT_NAME "${product}"
  FORMATS ${formats} IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics AU_MAIN_TYPE kAudioUnitType_Effect
  MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Optional input for processing. Input is muted on first launch.")
 target_sources(${target} PRIVATE src/plugin/ParameterManager.cpp src/plugin/PluginProcessor.cpp src/plugin/PluginFactory.cpp src/ui/PluginEditor.cpp)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 AURUM_COMPACT=${compact})
 target_include_directories(${target} PUBLIC src)
 target_link_libraries(${target} PUBLIC aurum_dsp PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
endfunction()

 set_tests_properties(host-foundation PROPERTIES ENVIRONMENT "AURUM_NO_SETTINGS=1")
endif()
