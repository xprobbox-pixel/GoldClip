cmake_minimum_required(VERSION 3.24)
include(cmake/Identity.cmake)
project(AurumClip VERSION ${AURUM_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(aurum_dsp STATIC src/dsp/ProcessingEngine.cpp)
target_include_directories(aurum_dsp PUBLIC src)
add_executable(AurumDspTests tests/DspTests.cpp)
target_link_libraries(AurumDspTests PRIVATE aurum_dsp)
add_test(NAME dsp-foundation COMMAND AurumDspTests)
